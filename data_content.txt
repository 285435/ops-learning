        {todos.map(t => <li key={t.id}>{t.text}</li>)}
      </ul>
    </div>
  )
}

package main

import (
    "fmt"
    "sync"
    "time"
)

func worker(id int, jobs <-chan int, results chan<- int, wg *sync.WaitGroup) {
    defer wg.Done()
    for j := range jobs {
        time.Sleep(100 * time.Millisecond)
        results <- j * j
    }
}

func main() {
    jobs := make(chan int, 10)
    results := make(chan int, 10)
    var wg sync.WaitGroup

    for w := 1; w <= 3; w++ {
        wg.Add(1)
        go worker(w, jobs, results, &wg)
    }

    for j := 1; j <= 5; j++ { jobs <- j }
    close(jobs)

    go func() { wg.Wait(); close(results) }()

    for r := range results {
        fmt.Println("结果:", r)
    }
}

// WebSocket 服务端实例 (Node.js)
const { WebSocketServer } = require('ws')
const wss = new WebSocketServer({ port: 8080 })
const clients = new Set()

wss.on('connection', (ws, req) => {
  const clientId = Date.now()
  ws.clientId = clientId
  clients.add(ws)

  // 广播消息
  ws.on('message', (data) => {
    const msg = JSON.parse(data)
    const broadcast = JSON.stringify({
      from: clientId,
      text: msg.text,
      time: new Date().toISOString()
    })
    clients.forEach(client => {
      if (client.readyState === 1) client.send(broadcast)
    })
  })

  ws.on('pong', () => { ws.isAlive = true })

  ws.on('close', () => {
    clients.delete(ws)
    console.log(`客户端 ${clientId} 断开`)
  })
})

// 心跳检测
setInterval(() => {
  clients.forEach(ws => {
    if (!ws.isAlive) return ws.terminate()
    ws.isAlive = false
    ws.ping()
  })
}, 30000)

cut 截取：-d: 分隔符、-f1 第1列、-c1-5 字符范围
tr 替换/删除：tr 'a-z' 'A-Z' 大写、tr -d ' ' 删除空格
tee 分流：同时输出到屏幕和文件
paste 合并文件：按列合并
diff 比较：-u 统一格式
wc 统计：-l 行数、-w 单词数、-c 字符数
xargs 参数传递：-I{} 替换、-P4 并行
EXAMPLE:
#!/bin/bash
# 文本处理工具集实例

# sort + uniq 统计
sort access.log | uniq -c | sort -rn | head -10
# 统计访问最多的前10个IP

# cut 截取
cut -d: -f1 /etc/passwd              # 所有用户名
cut -d' ' -f1-3 /var/log/nginx.log   # 日志前3列

# tr 替换/删除
echo "Hello World" | tr 'a-z' 'A-Z'  # HELLO WORLD
echo "a b c" | tr -d ' '             # abc

# tee 分流
echo "log entry" | tee -a app.log    # 同时输出和追加文件

# diff 比较
diff -u old.conf new.conf
diff <(ls dir1) <(ls dir2)

# comm 交集差集（需排序）
sort file1.txt > s1.txt
sort file2.txt > s2.txt
comm -12 s1.txt s2.txt   # 交集
comm -23 s1.txt s2.txt   # 只在file1中

# split 分割大文件
split -l 1000 large.log chunk_

# xargs 参数传递
find /src -name "*.py" | xargs grep "import" | wc -l
echo "1 2 3 4" | xargs -n1 echo     # 每行一个

# 综合管道：Nginx Top10 IP
awk '{print $1}' /var/log/nginx/access.log \
  | sort | uniq -c | sort -rn | head -10

TITLE: Shell 子shell与进程管理
LEVEL: 进阶
CONTENT:
子shell：(命令) 在新进程中执行，不影响当前shell变量
命令分组：{ 命令; } 在当前shell中执行

进程替换：
- <(command) 输出作为临时文件供读取
- >(command) 输入作为临时文件供写入

后台执行：
- command & 后台运行
- nohup command & 不挂断（终端关闭后继续）

作业控制：
- jobs 查看后台作业
- fg %n 调到前台、bg %n 后台继续
- wait 等待所有后台作业
- wait $PID 等待指定进程

命名管道：mkfifo 创建FIFO文件
EXAMPLE:
#!/bin/bash
# 子shell与进程管理实例

# === 子shell vs 命令分组 ===
count=0

# 子shell：变量修改不影响父shell
( count=100; echo "子shell: count=$count" )
echo "父shell: count=$count"      # 还是 0

# 命令分组：在当前shell执行
{ count=100; echo "分组: count=$count"; }
echo "分组后: count=$count"        # 变成 100

# === 后台执行 ===
sleep 10 &
echo "后台PID: $!"

nohup python app.py > app.log 2>&1 &
echo "nohup PID: $!"

# === wait 等待 ===
task1() { sleep 2; echo "任务1完成"; }
task2() { sleep 3; echo "任务2完成"; }

task1 &
task2 &
wait                    # 等待所有后台任务
echo "所有任务完成"

# === 进程替换 ===
diff <(ls /dir1) <(ls /dir2)

# 同时写入多个文件
echo "data" | tee >(gzip > out.gz) >(wc -c > count.txt) > /dev/null

# === 命名管道 ===
pipe=/tmp/my_pipe
mkfifo "$pipe"
# 终端1: echo "hello" > "$pipe"
# 终端2: cat < "$pipe"
rm -f "$pipe"
